function updateUndoToastProgress(){
 if(!pendingDeleteUndo)return;
 const remaining=Math.max(0,pendingDeleteUndo.expiresAt-Date.now());
 if(!remaining){clearPendingDeleteUndo();return;}
 const root=document.getElementById('toast-root');
 const toast=root?.querySelector('.toast-undo');
 if(!toast)return;
 const ratio=Math.max(0,Math.min(1,remaining/UNDO_DELETE_WINDOW_MS));
 const bar=toast.querySelector('[data-undo-progress]');
 if(bar)bar.style.width=`${(ratio*100).toFixed(2)}%`;
 const seconds=toast.querySelector('[data-undo-seconds]');
 if(seconds){
  const next=Math.ceil(remaining/1000);
  if(seconds.textContent!==String(next))seconds.textContent=String(next);
 }
}
function renderUndoToast(message='已删除内容',announce=false){
 const root=document.getElementById('toast-root');if(!root||!pendingDeleteUndo)return;
 const remaining=Math.max(0,pendingDeleteUndo.expiresAt-Date.now());
 if(!remaining){clearPendingDeleteUndo();return;}
 const ratio=Math.max(0,Math.min(1,remaining/UNDO_DELETE_WINDOW_MS));
 root.innerHTML=`<div class="toast toast-undo" role="status" aria-live="polite"><span class="toast-undo-icon" aria-hidden="true">↶</span><span class="toast-undo-copy"><strong>${esc(message)}</strong><small>${esc(pendingDeleteUndo.label)} · <span data-undo-seconds>${Math.ceil(remaining/1000)}</span> 秒内可撤销 · Ctrl/⌘Z</small></span><button type="button" class="toast-undo-action" data-action="undo-delete" aria-keyshortcuts="Control+Z Meta+Z">撤销</button><span class="toast-undo-progress" aria-hidden="true"><span data-undo-progress style="width:${(ratio*100).toFixed(2)}%"></span></span></div>`;
 if(undoDeleteTimer)clearTimeout(undoDeleteTimer);

}

function restorePendingDeleteUndo(){
 try{
  const raw=sessionStorage.getItem(UNDO_STORAGE_KEY);if(!raw)return false;
  const candidate=JSON.parse(raw);
  if(!candidate||typeof candidate.snapshot!=='string'||!Number.isFinite(candidate.expiresAt)||candidate.expiresAt<=Date.now()){sessionStorage.removeItem(UNDO_STORAGE_KEY);return false;}
  pendingDeleteUndo=candidate;return true;
 }catch(_){try{sessionStorage.removeItem(UNDO_STORAGE_KEY);}catch(__){}return false;}
}
